\intobmk\chapter*{符号列表}% 显示在书签但不显示在目录

\section*{物理量与量子数}
\nomenclatureitem{$I, M$}{总角动量量子数及其在实验室系 $z$ 轴的投影分量}
\nomenclatureitem{$\pi$}{空间宇称量子数（$\pi = \pm 1$）}
\nomenclatureitem{$K, \Omega$}{总角动量与单粒子角动量在内禀对称轴上的投影分量}
\nomenclatureitem{$\beta, \gamma$}{Bohr–Mottelson 四极形变参数（轴对称形变度与非对称偏角）}
\nomenclatureitem{$\mathcal{J}^{(1)}, \mathcal{J}^{(2)}$}{运动学转动惯量与动力学转动惯量}
\nomenclatureitem{$B(E\lambda), B(M\lambda)$}{约化电多极与磁多极跃迁几率}
\nomenclatureitem{$\delta$}{电磁跃迁多极混合比（如 $E2/M1$）}
\nomenclatureitem{$\Delta$}{BCS 配对超流能隙参数}
\nomenclatureitem{$u_k, v_k$}{Bogoliubov–Valatin 变换空穴振幅与占据振幅}
\nomenclatureitem{$\tau$}{激发态平均寿命}
\nomenclatureitem{$\alpha_\mathrm{tot}, \alpha_K$}{总内转换系数与 $K$ 壳层内转换系数}

\section*{数学算符与张量记号}
\nomenclatureitem{$\hat{H}$}{核多体哈密顿量算符}
\nomenclatureitem{$\hat{P}^I_{MK}$}{三维角动量投影算符}
\nomenclatureitem{$\hat{T}_{\lambda\mu}$}{不可约球张量多极跃迁算符}
\nomenclatureitem{$D^I_{MK}(\Omega)$}{Wigner $D$ 矩阵旋转表示矩阵元}
\nomenclatureitem{$\langle I_i K_i \lambda \mu | I_f K_f \rangle$}{Clebsch–Gordan 矢量耦合系数}
\nomenclatureitem{$\mathcal{C}$}{反幺正时间反演与内禀旋转联合手征变换算符}

\section*{主要缩略语}
\nomenclatureitem{BCS}{Bardeen–Cooper–Schrieffer（超流配对理论）}
\nomenclatureitem{CFD}{Constant Fraction Discriminator（恒定比定时甄别器）}
\nomenclatureitem{DPP}{Digital Pulse Processing（数字脉冲处理技术）}
\nomenclatureitem{DSAM}{Doppler Shift Attenuation Method（多普勒频移衰减法）}
\nomenclatureitem{ESPE}{Effective Single-Particle Energy（有效单粒子能）}
\nomenclatureitem{HPGe}{High-Purity Germanium（高纯锗探测器）}
\nomenclatureitem{MCSM}{Monte Carlo Shell Model（蒙特卡罗壳模型）}
\nomenclatureitem{PRM}{Particle-Rotor Model（粒子—转子耦合模型）}
\nomenclatureitem{RDDS}{Recoil Distance Doppler Shift（反冲距离多普勒频移法 / 柱塞法）}
\nomenclatureitem{RME}{Reduced Matrix Element（约化矩阵元）}
\nomenclatureitem{TAC}{Tilted Axis Cranking（倾斜轴曲轴转动模型）}
\nomenclatureitem{TSD}{Triaxial Strongly Deformed（三轴强形变带）}
