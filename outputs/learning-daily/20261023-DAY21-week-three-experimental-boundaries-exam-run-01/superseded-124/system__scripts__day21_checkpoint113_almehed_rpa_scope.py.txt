#!/usr/bin/env python3
"""Almehed 2011 TAC+RPA small-amplitude expansion, quadrupole inversion Jacobian and softening scope."""
import json
import math
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
RUN = ROOT / "outputs/learning-daily/20261023-DAY21-week-three-experimental-boundaries-exam-run-01"

def audit():
    # 1. Linear vs exact finite rotation for R3 = exp(-i phi J3)
    # Linear expansion: Delta Q2- / <Q2+> = 2 phi
    # Exact finite rotation: Delta Q2- / <Q2+> = sin(2 phi)
    angles_deg = [10.0, 20.0, 30.0, 40.0, 45.0]
    expansion_results = []
    for a in angles_deg:
        phi_rad = math.radians(a)
        linear = 2.0 * phi_rad
        exact = math.sin(2.0 * phi_rad)
        rel_diff = (linear - exact) / exact
        expansion_results.append({
            "angle_deg": a,
            "phi_rad": phi_rad,
            "linear_2phi": linear,
            "exact_sin2phi": exact,
            "relative_overestimation_pct": rel_diff * 100.0
        })

    # 2. Quadrupole coordinate inversion Jacobian denominator
    # Denominator in Eqs.(A5)-(A6): <Q0>^2 + <Q2+>^2 = C^2 epsilon^2 (cos^2 gamma + sin^2 gamma) = C^2 epsilon^2
    return {
        "scope": "Almehed 2011 TAC+RPA small-amplitude expansion and harmonic approximation validity limits",
        "source": {
            "doi": "10.1103/PhysRevC.83.054308",
            "arxiv": "0709.0969v2",
            "locators": ["PRC 83, 054308 p.2", "p.4", "p.8", "p.17-18 Appendix A Eqs.(A3)-(A10)"]
        },
        "small_amplitude_anharmonicity_check": expansion_results,
        "quadrupole_inversion_denominator": {
            "formula": "<Q0>^2 + <Q2+>^2 = C^2 * epsilon^2",
            "spherical_limit_singularity": "As epsilon -> 0 (spherical nucleus), the coordinate inversion Jacobian diverges, reflecting coordinate singularity of deformation angles."
        },
        "author_stated_limitations": [
            "The RPA accounts for harmonic excitations above the mean field minimum; it breaks down when phonon energy decreases toward zero near the critical point.",
            "The calculated RPA phonon energy tends to approach zero faster than in experiment, proving the necessity of large-amplitude anharmonic treatments.",
            "135Nd calculation was assigned to Mukhopadhyay 2007 (Ref.13) and explicitly stated as 'not shown here' (p.8); odd-odd neighbor parameters cannot be substituted as 135Nd inputs."
        ],
        "inference_boundary": [
            "RPA phonon energy softening to zero is an artifact of the breakdown of small-amplitude harmonic approximations, not experimental proof of a static phase boundary.",
            "Neither TAC+RPA nor fixed-deformation PRM can be treated as closed parameter-free microscopic predictions for 135Nd.",
            "A genuine chiral falsification test requires freezing a specific model parameter domain Theta rather than switching between models post-hoc."
        ],
        "raw_events_acquired": False,
        "rpa_code_replayed": False,
        "new_CI": None,
        "actual_joint_covariance": None,
        "scientific_certification": False
    }

if __name__ == "__main__":
    d = audit()
    out = RUN / "almehed-rpa-scope-113.json"
    out.write_text(json.dumps(d, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps({
        "output": str(out.relative_to(ROOT)),
        "overest_30deg_pct": d["small_amplitude_anharmonicity_check"][2]["relative_overestimation_pct"],
        "overest_40deg_pct": d["small_amplitude_anharmonicity_check"][3]["relative_overestimation_pct"]
    }, ensure_ascii=False))
