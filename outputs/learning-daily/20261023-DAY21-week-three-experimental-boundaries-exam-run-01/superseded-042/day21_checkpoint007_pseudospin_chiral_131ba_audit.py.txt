def compute_quartet_dispersion():
    spins = [11.5, 12.5, 13.5, 14.5, 15.5, 16.5, 17.5]
    results = []
    for I in spins:
        spin_label = f"{int(2*I)}/2+"
        e_vals = {}
        for b in ["D3", "D4", "D5", "D6"]:
            if I in bands_data[b]:
                e_vals[b] = bands_data[b][I]
        min_b = min(e_vals, key=e_vals.get)
        max_b = max(e_vals, key=e_vals.get)
        disp = e_vals[max_b] - e_vals[min_b]

        # Sort bands by energy
        sorted_order = sorted(e_vals.keys(), key=lambda k: e_vals[k])

        results.append({
            "spin": spin_label,
            "I": I,
            "energies_keV": e_vals,
            "energy_ordering": sorted_order,
            "lowest_band": min_b,
            "highest_band": max_b,
            "energy_dispersion_keV": round(disp, 1),
            "d4_d5_gap_keV": round(e_vals["D5"] - e_vals["D4"], 1)
        })
    return results

def compute_staggering():
    results = {}
    for b, levels in bands_data.items():
        s_vals = []
        sorted_spins = sorted(levels.keys())
        for i in range(1, len(sorted_spins)):
            I = sorted_spins[i]
            I_prev = sorted_spins[i-1]
            if abs(I - I_prev - 1.0) < 1e-4:
                dE = levels[I] - levels[I_prev]
                S = dE / (2.0 * I)
                s_vals.append({
                    "spin": f"{int(2*I)}/2+",
                    "I": I,
                    "delta_E_keV": round(dE, 1),
                    "S_I_keV": round(S, 2)
                })
        results[b] = s_vals
    return results

def branching_sensitivity():
    # Test formula: B(M1)/B(E2) = 0.697 * (E_gamma(E2)^5 / E_gamma(M1)^3) * (1 / lambda) * (1 / (1 + delta^2))
    # Let E_M1 = 300 keV, E_E2 = 600 keV, branching ratio lambda = I_gamma(E2)/I_gamma(M1) = 0.1
    # Vary delta from 0.0 to -0.2 and -0.3
    E_M1 = 300.0 # keV
    E_E2 = 600.0 # keV
    lam = 0.1

    deltas = [0.0, -0.1, -0.2, -0.3]
    res = []
    base_val = None
    for d in deltas:
        val = 0.697 * ((E_E2 / 1000.0) ** 5) / ((E_M1 / 1000.0) ** 3) * (1.0 / lam) * (1.0 / (1.0 + d**2))
        if base_val is None:
            base_val = val
        shift_pct = (val - base_val) / base_val * 100.0
        res.append({
            "assumed_delta": d,
            "bm1_be2_ratio_muN2_e2b2": round(val, 3),
            "relative_shift_percent": round(shift_pct, 2)
        })
    return res
