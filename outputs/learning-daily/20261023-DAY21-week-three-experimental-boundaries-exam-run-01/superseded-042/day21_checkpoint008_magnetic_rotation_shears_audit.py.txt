import json
import math
from pathlib import Path

def compute_shears_kinematics(j1, j2, spins):
    # Semiclassical shears formula
    # cos(theta) = (I^2 - j1^2 - j2^2) / (2 * j1 * j2)
    # sin^2(theta) = 1 - cos^2(theta)
    # B(M1) normalized to bandhead maximum
    results = []
    for I in spins:
        cos_th = (I**2 - j1**2 - j2**2) / (2.0 * j1 * j2)
        # bound to [-1, 1]
        cos_th_clamped = max(-1.0, min(1.0, cos_th))
        theta_rad = math.acos(cos_th_clamped)
        theta_deg = math.degrees(theta_rad)
        sin2_th = math.sin(theta_rad) ** 2
        results.append({
            "I": I,
            "cos_theta": round(cos_th_clamped, 4),
            "theta_deg": round(theta_deg, 2),
            "sin2_theta": round(sin2_th, 4)
        })
    return results
