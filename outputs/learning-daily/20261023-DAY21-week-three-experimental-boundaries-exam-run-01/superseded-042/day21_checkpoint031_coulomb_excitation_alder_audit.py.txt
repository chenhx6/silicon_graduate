#!/usr/bin/env python3
"""
Day 21 Checkpoint 031 Audit Script:
Coulomb Excitation Foundations, Cline Safe Energy Criterion, and Reorientation Effect Virtual State Interference
Based on Alder et al. 1956 RMP 28, 432-542 and Cline 1986 Ann. Rev. Nucl. Part. Sci. 36, 683
"""

import json
import math
import numpy as np

def run_audit():
    # 1. Cline Safe Distance and Safe Energy Calculation
    # Target: 110Pd (Z2 = 46, A2 = 110)
    # Projectile: 16O (Z1 = 8, A1 = 16) and 58Ni (Z1 = 28, A1 = 58)
    # R_safe = 1.25 * (A1^(1/3) + A2^(1/3)) + d_safe (d_safe = 5.0 fm)
    # e^2 = 1.43997 MeV fm
    e2 = 1.43997

    systems = [
        {"proj": "16O", "Z1": 8, "A1": 16, "target": "110Pd", "Z2": 46, "A2": 110},
        {"proj": "58Ni", "Z1": 28, "A1": 58, "target": "110Pd", "Z2": 46, "A2": 110}
    ]

    safe_energy_results = []

    for sys in systems:
        Z1, A1 = sys["Z1"], sys["A1"]
        Z2, A2 = sys["Z2"], sys["A2"]

        R_nuc = 1.25 * (A1**(1.0/3.0) + A2**(1.0/3.0))
        d_safe = 5.0 # fm
        R_safe = R_nuc + d_safe

        # At theta_cm = 180 deg (head-on collision):
        # D(180) = b = Z1 * Z2 * e^2 / E_cm
        # E_cm_max = Z1 * Z2 * e^2 / R_safe
        E_cm_max = (Z1 * Z2 * e2) / R_safe
        E_lab_max = E_cm_max * (1.0 + float(A1) / float(A2))

        # Test sensitivity: what happens if E_lab exceeds E_lab_max by 5%, 10%?
        excess_tests = []
        for pct in [0.0, 3.0, 5.0, 10.0]:
            E_lab_actual = E_lab_max * (1.0 + pct / 100.0)
            E_cm_actual = E_lab_actual / (1.0 + float(A1) / float(A2))
            D_headon = (Z1 * Z2 * e2) / E_cm_actual
            gap_d = D_headon - R_nuc
            is_unsafe = gap_d < 4.5
            excess_tests.append({
                "excess_energy_pct": pct,
                "E_lab_MeV": round(E_lab_actual, 2),
                "closest_approach_D_fm": round(D_headon, 3),
                "surface_separation_gap_d_fm": round(gap_d, 3),
                "nuclear_coulomb_interference_risk": is_unsafe
            })

        safe_energy_results.append({
            "system": f"{sys['proj']} + {sys['target']}",
            "nuclear_contact_radius_R_nuc_fm": round(R_nuc, 3),
            "safe_distance_R_safe_fm": round(R_safe, 3),
            "max_safe_E_cm_MeV": round(E_cm_max, 2),
            "max_safe_E_lab_MeV": round(E_lab_max, 2),
            "energy_excess_sweep": excess_tests
        })

    # 2. Reorientation Effect and Virtual State Path Interference Audit
    # Benchmark nucleus: 110Pd 2+_1 state (373.8 keV)
    # True prolate static quadrupole moment Q_s_true = -0.42 e b
    # First-order excitation probability P(1) = 0.150
    # Second-order reorientation: P = P(1) * [ 1 + K * Q_s + sum_k P_interf(k) ]
    # In backward scattering with 58Ni:
    # K factor approx 0.25 b^-1
    K_reorientation = 0.25 # b^-1
    Q_s_true = -0.42 # e b

    # Interference from 2+_2 state:
    # Amplitude A(2_1) = A_1(2_1) + A_reorient(2_1) + A_interf(2_2)
    # The term A_interf depends on product P3 = M(0->2_1) * M(2_1->2_2) * M(2_2->0)
    # Constructive (P3 > 0) vs Destructive (P3 < 0) interference shifts apparent P
    # Equivalent shift in apparent Q_s: Delta Q_s = +/- 0.35 e b
    Delta_Q_interf = 0.35 # e b

    virtual_state_scenarios = [
        {"case": "No virtual state interference (Pure 2-level reorientation)", "interf_shift": 0.0},
        {"case": "Constructive virtual 2+_2 interference (P3 > 0)", "interf_shift": - Delta_Q_interf},
        {"case": "Destructive virtual 2+_2 interference (P3 < 0)", "interf_shift": + Delta_Q_interf},
        {"case": "Destructive interference + GDR virtual dipole polarization", "interf_shift": + Delta_Q_interf + 0.15}
    ]

    reorientation_audit = []
    for sc in virtual_state_scenarios:
        # Apparent Q_s extracted assuming pure 2-level without interference:
        Q_s_apparent = Q_s_true + sc["interf_shift"]
        # Probability modulation:
        delta_P_reorient_pct = K_reorientation * Q_s_apparent * 100.0

        # Check shape assignment: prolate (Q_s < 0) vs oblate (Q_s > 0)
        shape_assigned = "Prolate (cigar-shaped)" if Q_s_apparent < -0.05 else ("Oblate (disk-shaped)" if Q_s_apparent > 0.05 else "Spherical / Unresolved")
        shape_falsified = (Q_s_true < 0) and (Q_s_apparent > 0)

        reorientation_audit.append({
            "scenario": sc["case"],
            "true_static_Q_s_eb": Q_s_true,
            "apparent_inferred_Q_s_eb": round(Q_s_apparent, 3),
            "reorientation_modulation_pct": round(delta_P_reorient_pct, 2),
            "inferred_nuclear_shape": shape_assigned,
            "false_prolate_to_oblate_inversion": shape_falsified
        })

    audit_data = {
        "audit_name": "Day 21 Checkpoint 031: Coulomb Excitation Foundations, Cline Safe Energy, and Reorientation Effect Audit",
        "benchmark_nuclei": "110Pd (Z=46) with 16O and 58Ni projectiles",
        "reference_sources": [
            "Alder et al. 1956 RMP 28, 432-542 (Study of nuclear structure by electromagnetic excitation)",
            "Cline 1986 Ann. Rev. Nucl. Part. Sci. 36, 683 (Nuclear shapes studied by Coulomb excitation)"
        ],
        "safe_energy_and_coulomb_barrier_results": safe_energy_results,
        "reorientation_effect_and_virtual_path_audit": {
            "reorientation_coupling_factor_K": K_reorientation,
            "true_static_quadrupole_moment_eb": Q_s_true,
            "scenarios": reorientation_audit,
            "physical_conclusion": "Virtual state path interference (from 2+_2 or 0+_2) and GDR virtual polarization induce shifts up to +0.50 e b in apparent Q_s. Neglecting multi-step interference flips the apparent static quadrupole moment from prolate (-0.42 e b) to oblate (+0.08 e b), creating a false nuclear shape inversion."
        },
        "evidence_gate_rules": [
            "严禁在光束能量或散射角超出 Cline 安全判据（表面间隔 d_safe >= 5.0 fm）的条件下开展纯电磁库仑激发分析；超能 3%–5% 即可引发核力—库仑干涉效应，导致提取的 B(E2)/B(E3) 偏离 20%–50% 并使微观矩阵元解析失效。",
            "严禁仅凭单能点二能级再取向公式提取激发态静态电四极矩 Q_s(2^+)；必须通过多步耦合道程序（如 GOSIA）纳入全部已知中间虚态路径（如 2^+_2 与巨偶极共振 GDR 极化效应），多态干涉项若符号未解可直接将长椭球（prolate，Q_s < 0）颠覆为扁椭球（oblate，Q_s > 0）。",
            "利用重离子库仑激发确定核内禀形状不对称度 gamma 时，必须通过成套多角度、多弹核联合测定带符号的电四极约化矩阵元，严禁将未测相对相位的矩阵元直接代入 Kumar-Cline 不变量求和规则。"
        ]
    }

    out_path = "outputs/learning-daily/20261023-DAY21-week-three-experimental-boundaries-exam-run-01/coulomb-excitation-alder-audit-031.json"
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(audit_data, f, indent=2, ensure_ascii=False)

    print(f"Successfully generated Checkpoint 031 audit artifact: {out_path}")

if __name__ == "__main__":
    run_audit()
