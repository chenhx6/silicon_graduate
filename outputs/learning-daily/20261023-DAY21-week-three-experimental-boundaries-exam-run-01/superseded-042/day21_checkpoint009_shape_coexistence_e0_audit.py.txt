import json
import math

def two_state_e0_model(Z, A, delta_E_keV, V_mix_keV, beta_norm, beta_def):
    # delta_E = sqrt(delta_0^2 + 4 * V^2)
    # If delta_E and V are given, unperturbed gap delta_0 = sqrt(delta_E^2 - 4 * V^2)
    delta_0_sq = delta_E_keV**2 - 4.0 * (V_mix_keV**2)
    delta_0_keV = math.sqrt(max(0.0, delta_0_sq))

    # Mixing probabilities: a^2 * b^2 = V^2 / delta_E^2
    a2_b2 = (V_mix_keV / delta_E_keV) ** 2
    b2 = 0.5 * (1.0 - math.sqrt(max(0.0, 1.0 - 4.0 * a2_b2)))
    a2 = 1.0 - b2

    # E0 strength: rho^2(E0) = (3 / (4 * pi) * Z)^2 * a2_b2 * (beta_def^2 - beta_norm^2)^2
    factor_geom = (3.0 / (4.0 * math.pi) * Z) ** 2
    delta_beta2_sq = (beta_def**2 - beta_norm**2) ** 2
    rho2_e0 = factor_geom * a2_b2 * delta_beta2_sq
    rho2_e0_milli = rho2_e0 * 1000.0 # in units of 10^-3

    return {
        "Z": Z,
        "A": A,
        "delta_E_observed_keV": delta_E_keV,
        "V_mix_keV": V_mix_keV,
        "unperturbed_gap_delta0_keV": round(delta_0_keV, 1),
        "mixing_weight_a2": round(a2, 4),
        "mixing_weight_b2": round(b2, 4),
        "mixing_product_a2_b2": round(a2_b2, 5),
        "beta_normal": beta_norm,
        "beta_deformed": beta_def,
        "delta_beta2": round(beta_def**2 - beta_norm**2, 4),
        "rho2_e0": round(rho2_e0, 6),
        "rho2_e0_x1000": round(rho2_e0_milli, 2)
    }
