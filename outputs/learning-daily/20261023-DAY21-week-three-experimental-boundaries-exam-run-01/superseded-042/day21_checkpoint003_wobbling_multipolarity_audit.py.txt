"""
Day 21 Checkpoint 003 Quantitative Audit:

import json
import math
from pathlib import Path

def compute_fractions(delta):
    delta_sq = delta ** 2
    q_e2 = delta_sq / (1.0 + delta_sq)
    q_m1 = 1.0 / (1.0 + delta_sq)
    return round(q_e2 * 100.0, 2), round(q_m1 * 100.0, 2)

def audit_135pr_transitions():
    # Experimental data from Lv et al. 2022 (PLB 824, 136840, Table 1 & text)
    transitions = [
        {
            "energy_kev": 747.3,
            "initial_state": "17/2- (Band 3 / former TW1)",
            "final_state": "15/2- (Band 1 / yrast)",
            "delta_lv2022": -0.47,
            "delta_err_plus": 0.09,
            "delta_err_minus": 0.22,
            "delta_matta2015_claimed": -1.24,  # or +2.1
            "pol_asymmetry_P": 0.04,
            "R_ac": 0.50
        },
        {
            "energy_kev": 813.2,
            "initial_state": "21/2- (Band 3 / former TW1)",
            "final_state": "19/2- (Band 1 / yrast)",
            "delta_lv2022": -0.37,
            "delta_err_plus": 0.10,
            "delta_err_minus": 0.14,
            "delta_matta2015_claimed": -1.54,
            "pol_asymmetry_P": -0.03,
            "R_ac": 0.48
        },
        {
            "energy_kev": 450.2,
            "initial_state": "19/2- (former TW2 candidate)",
            "final_state": "17/2- (Band 3)",
            "delta_lv2022": -0.31,
            "delta_err_plus": 0.10,
            "delta_err_minus": 0.13,
            "delta_matta2015_claimed": None,  # Sensharma 2019 claimed wobbling link
            "pol_asymmetry_P": -0.07,
            "R_ac": 0.44
        }
    ]

    results = []
    for t in transitions:
        qe2_lv, qm1_lv = compute_fractions(t["delta_lv2022"])
        row = {
            "energy_kev": t["energy_kev"],
            "states": f"{t['initial_state']} -> {t['final_state']}",
            "delta_lv2022": t["delta_lv2022"],
            "E2_percent_lv2022": qe2_lv,
            "M1_percent_lv2022": qm1_lv,
            "is_M1_dominated": qm1_lv > 50.0,
            "m1_dominance_ratio": round(qm1_lv / qe2_lv, 2)
        }
        if t["delta_matta2015_claimed"] is not None:
            qe2_matta, qm1_matta = compute_fractions(t["delta_matta2015_claimed"])
            row["delta_matta2015"] = t["delta_matta2015_claimed"]
            row["E2_percent_matta2015"] = qe2_matta
            row["M1_percent_matta2015"] = qm1_matta
            row["claimed_E2_dominated"] = qe2_matta > 50.0
        results.append(row)

    return results
