6. PES barrier vs zero-point energy (ZPE) evaluation for static pear shape vs octupole softness.

def factorial(n):
    return math.factorial(n)

def clebsch_gordan(j1, j2, j3, m1, m2, m3):
    if m1 + m2 != m3:
        return 0.0
    if not (abs(j1 - j2) <= j3 <= j1 + j2):
        return 0.0
    f = math.factorial
    factor = math.sqrt((2 * j3 + 1) * f(j1 + j2 - j3) * f(j1 - j2 + j3) * f(-j1 + j2 + j3) / f(j1 + j2 + j3 + 1))
    factor *= math.sqrt(f(j1 + m1) * f(j1 - m1) * f(j2 + m2) * f(j2 - m2) * f(j3 + m3) * f(j3 - m3))
    s_min = max(0, j2 - m1 - j3, j1 + m2 - j3)
    s_max = min(j1 + j2 - j3, j1 - m1, j2 + m2)
    total = 0.0
    for s in range(s_min, s_max + 1):
        num = ((-1) ** s)
        den = f(s) * f(j1 + j2 - j3 - s) * f(j1 - m1 - s) * f(j2 + m2 - s) * f(j3 - j2 + m1 + s) * f(j3 - j1 - m2 + s)
        total += num / den
    return factor * total

def compute_be3_wu(A, me_val):
    # me_val in e * b^(3/2)
    # B(E3; 0+ -> 3-) = |me|^2 (in e^2 b^3)
    # B(E3; 3- -> 0+) = 1/7 * B(E3; 0+ -> 3-)
    # 1 b = 100 fm^2, 1 b^3 = 10^6 fm^6
    R0 = 1.2 * (A ** (1.0 / 3.0))
    b_wu_fm6 = (1.0 / (4.0 * math.pi)) * ((3.0 / 6.0) ** 2) * (R0 ** 6)

    be3_0_to_3_fm6 = (me_val ** 2) * 1e6
    be3_3_to_0_fm6 = be3_0_to_3_fm6 / 7.0
    be3_wu = be3_3_to_0_fm6 / b_wu_fm6
    return {
        "A": A,
        "R0_fm": round(R0, 4),
        "B_Wu_fm6": round(b_wu_fm6, 2),
        "me_eb3_2": me_val,
        "be3_3_to_0_fm6": round(be3_3_to_0_fm6, 1),
        "be3_wu": round(be3_wu, 2)
    }

def compute_q3_beta3(A, Z, me_val, beta2=0.18):
    # Leander & Chen 1988 formulation:
    # Q3 = sqrt(16 pi / 7) * me (in e fm^3)
    # Q3 = 3 / sqrt(7 pi) * Z * e * R0^3 * [ beta3 + (9 / (7 sqrt(pi))) * beta2 * beta3 ]
    R0 = 1.2 * (A ** (1.0 / 3.0))
    factor_q3 = math.sqrt(16.0 * math.pi / 7.0)
    me_efm3 = me_val * 1000.0 # 1 b^(3/2) = 1000 fm^3
    Q3 = factor_q3 * me_efm3

    geom_factor = (3.0 / math.sqrt(7.0 * math.pi)) * Z * (R0 ** 3)
    corr_factor = 1.0 + (9.0 / (7.0 * math.sqrt(math.pi))) * beta2
    beta3 = Q3 / (geom_factor * corr_factor)
    return {
        "A": A,
        "Z": Z,
        "me_eb3_2": me_val,
        "Q3_efm3": round(Q3, 1),
        "geom_factor": round(geom_factor, 1),
        "beta2": beta2,
        "corr_factor": round(corr_factor, 4),
        "beta3": round(beta3, 4)
    }

def audit_rotor_predictions(me_0_to_3):
    transitions = [
        ("0+ -> 3-", 0, 3),
        ("2+ -> 5-", 2, 5),
        ("4+ -> 7-", 4, 7),
        ("6+ -> 9-", 6, 9),
    ]
    results = []
    for label, Ii, If in transitions:
        cg = clebsch_gordan(Ii, 3, If, 0, 0, 0)
        ratio = math.sqrt(2 * Ii + 1) * cg
        m_pred = me_0_to_3 * ratio
        results.append({
            "transition": label,
            "Ii": Ii,
            "If": If,
            "cg_coeff": round(cg, 4),
            "rotor_ratio": round(ratio, 4),
            "predicted_me_eb3_2": round(m_pred, 3)
        })
    return results

def compute_parity_splitting(bands):
    pos = dict(bands["pos"])
    results = []
    for I, E_neg in bands["neg"]:
        if (I - 1) in pos and (I + 1) in pos:
            E_pos_avg = 0.5 * (pos[I - 1] + pos[I + 1])
            delta_E = E_neg - E_pos_avg
            results.append({
                "I_minus": I,
                "E_neg_keV": E_neg,
                "E_pos_interpolated_keV": round(E_pos_avg, 1),
                "delta_E_keV": round(delta_E, 1)
            })
        elif (I - 1) in pos:
            delta_E = E_neg - pos[I - 1]
            results.append({
                "I_minus": I,
                "E_neg_keV": E_neg,
                "E_pos_interpolated_keV": round(pos[I - 1], 1),
                "delta_E_keV": round(delta_E, 1)
            })
    return results
